结构体模板 if_

boost::proto::if_ — 用于根据编译期布尔结果选择一个文法或另一个. 当用作变换时, proto::if_<> 根据编译期布尔结果在两个变换之间选择.

概要

// In header: <boost/proto/matches.hpp>

template<typename If, typename Then = proto::_, 
         typename Else = proto::not_<proto::_> > 
struct if_ :  proto::transform<if_<If, Then, Else> > {
  // types
  typedef if_ proto_grammar;

  // member classes/structs/unions
  template<typename Expr, typename State, typename Data> 
  struct impl :  proto::transform_impl< Expr, State, Data > {
    // types
    typedef typename mpl::if_<
      typename boost::result_of<proto::when<proto::_, If>(Expr, State, Data)>::type,
      typename boost::result_of<proto::when<proto::_, Then>(Expr, State, Data)>::type,
      typename boost::result_of<proto::when<proto::_, Else>(Expr, State, Data)>::type
    >::type result_type;

    // public member functions
    result_type operator()(typename impl::expr_param, 
                           typename impl::state_param, 
                           typename impl::data_param) const;
  };
};

说明

当 proto::if_<If, Then, Else> 被用作文法时, If 必须是一个 Proto 变换, Then 和 Else 必须是文法. 表达式类型 E 匹配 proto::if_<If, Then, Else>, 当 boost::result_of<proto::when<proto::_, If>(E)>::type::value 为 true 且 E 匹配 Then 时; 或者当 boost::result_of<proto::when<proto::_, If>(E)>::type::value 为 false 且 E 匹配 Else 时.

模板参数 Then 默认为 proto::_, Else 默认为 proto::not_<proto::_>, 因此表达式类型 E 将匹配 proto::if_<If>, 当且仅当 boost::result_of<proto::when<proto::_, If>(E)>::type::value 为 true.

// A grammar that only matches integral terminals,
// using is_integral<> from Boost.Type_traits.
struct IsIntegral :
  proto::and_<
    proto::terminal<proto::_>,
    proto::if_< boost::is_integral<proto::_value>()>
  >
{};

当 proto::if_<If, Then, Else> 被用作变换时, If、Then 和 Else 必须是 Proto 变换. 把该变换应用于表达式 E、状态 S 和数据 V 时, 如果 boost::result_of<proto::when<proto::_, If>(E,S,V)>::type::value 为 true, 则应用 Then 变换; 否则应用 Else 变换.

// Match a terminal. If the terminal is integral, return
// mpl::true_; otherwise, return mpl::false_.
struct IsIntegral2 :
  proto::when<
    proto::terminal<_>,
    proto::if_<
      boost::is_integral<proto::_value>(),
      mpl::true_(),
      mpl::false_()
    >
  >
{};
