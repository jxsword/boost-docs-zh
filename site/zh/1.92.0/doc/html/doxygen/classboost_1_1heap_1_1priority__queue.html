类模板 priority_queue

boost::heap::priority_queue — 基于 STL 堆函数的优先队列

概要

// In header: <boost/heap/priority_queue.hpp>

template<typename T, class... Options> 
class priority_queue {
public:
  // types
  typedef T                                       value_type;     
  typedef implementation_defined::size_type       size_type;      
  typedef implementation_defined::difference_type difference_type;
  typedef implementation_defined::value_compare   value_compare;  
  typedef implementation_defined::allocator_type  allocator_type; 
  typedef implementation_defined::reference       reference;      
  typedef implementation_defined::const_reference const_reference;
  typedef implementation_defined::pointer         pointer;        
  typedef implementation_defined::const_pointer   const_pointer;  
  typedef implementation_defined::iterator        iterator;       
  typedef implementation_defined::const_iterator  const_iterator; 

  // public member functions
  explicit priority_queue(value_compare const & = value_compare());
  explicit priority_queue(allocator_type const &);
  priority_queue(priority_queue const &) = default;
  priority_queue(priority_queue &&) = default;
  priority_queue & operator=(priority_queue &&) = default;
  priority_queue & operator=(priority_queue const &);
  bool empty(void) const noexcept;
  size_type size(void) const noexcept;
  size_type max_size(void) const noexcept;
  void clear(void) noexcept;
  allocator_type get_allocator(void) const;
  const_reference top(void) const;
  void push(value_type const &);
  template<class... Args> void emplace(Args &&...);
  void pop(void);
  void swap(priority_queue &);
  iterator begin(void) const noexcept;
  iterator end(void) const noexcept;
  void reserve(size_type);
  value_compare const & value_comp(void) const;
  template<typename HeapType> bool operator<(HeapType const &) const;
  template<typename HeapType> bool operator>(HeapType const &) const;
  template<typename HeapType> bool operator>=(HeapType const &) const;
  template<typename HeapType> bool operator<=(HeapType const &) const;
  template<typename HeapType> bool operator==(HeapType const &) const;
  template<typename HeapType> bool operator!=(HeapType const &) const;

  // private member functions
  void do_swap(priority_queue &);

  // public data members
  static const bool constant_time_size;
  static const bool has_ordered_iterators;
  static const bool is_mergable;
  static const bool is_stable;
  static const bool has_reserve;
};

说明

priority_queue 类是 STL 堆函数的封装.
模板参数 T 是容器要管理的类型. 用户可以指定附加选项; 若未提供选项, 则使用默认选项.

容器支持以下选项:

  • boost::heap::compare<>, 默认为 compare<std::less<T> >

  • boost::heap::stable<>, 默认为 stable<false>

  • boost::heap::stability_counter_type<>, 默认为 stability_counter_type<boost::uintmax_t>

  • boost::heap::allocator<>, 默认为 allocator<std::allocator<T> >

  1. explicit priority_queue(value_compare const & cmp = value_compare());

    效果: 构造一个空的优先队列.

    复杂度: 常数.

  2. explicit priority_queue(allocator_type const & alloc);

    效果: 构造一个空的优先队列.

    复杂度: 常数.

  3. priority_queue(priority_queue const & rhs) = default;

    效果: 从 rhs 复制构造优先队列.

    复杂度: 线性.

  4. priority_queue(priority_queue && rhs) = default;

    效果: C++11 风格的移动构造函数.

    复杂度: 常数.

  5. priority_queue & operator=(priority_queue && rhs) = default;

    效果: C++11 风格的移动赋值.

    复杂度: 常数.

  6. priority_queue & operator=(priority_queue const & rhs);

    效果: 从 rhs 赋值优先队列.

    复杂度: 线性.

  7. bool empty(void) const noexcept;

    效果: 若优先队列不包含任何元素, 则返回 true.

    复杂度: 常数.

  8. size_type size(void) const noexcept;

    效果: 返回优先队列中包含的元素数量.

    复杂度: 常数.

  9. size_type max_size(void) const noexcept;

    效果: 返回优先队列可包含的最大元素数量.

    复杂度: 常数.

  10. void clear(void) noexcept;

    效果: 从优先队列中移除所有元素.

    复杂度: 线性.

  11. allocator_type get_allocator(void) const;

    效果: 返回分配器.

    复杂度: 常数.

  12. const_reference top(void) const;

    效果: 返回最大元素的 const_reference.

    复杂度: 常数.

  13. void push(value_type const & v);

    效果: 向优先队列添加新元素.

    复杂度: 对数 (均摊). 线性 (最坏情况).

  14. template<class... Args> void emplace(Args &&... args);

    效果: 向优先队列添加新元素. 元素直接就地构造.

    复杂度: 对数 (均摊). 线性 (最坏情况).

  15. void pop(void);

    效果: 从优先队列移除顶部元素.

    复杂度: 对数 (均摊). 线性 (最坏情况).

  16. void swap(priority_queue & rhs);

    效果: 交换两个优先队列.
