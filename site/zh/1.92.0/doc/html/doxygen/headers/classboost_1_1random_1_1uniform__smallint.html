类模板 uniform_smallint

boost::random::uniform_smallint

概要

// In header: <boost/random/uniform_smallint.hpp>

template<typename IntType = int> 
class uniform_smallint {
public:
  // types
  typedef IntType input_type; 
  typedef IntType result_type;

  // member classes/structs/unions

  class param_type {
  public:
    // types
    typedef uniform_smallint distribution_type;

    // public member functions
    param_type(IntType = 0, IntType = 9);
    IntType a() const;
    IntType b() const;

    // friend functions
    template<typename CharT, typename Traits> 
      std::basic_ostream< CharT, Traits > & 
      operator<<(std::basic_ostream< CharT, Traits > &, const param_type &);
    template<typename CharT, typename Traits> 
      std::basic_istream< CharT, Traits > & 
      operator>>(std::basic_istream< CharT, Traits > &, const param_type &);
    bool operator==(const param_type &, const param_type &);
    bool operator!=(const param_type &, const param_type &);
  };

  // public member functions
  explicit uniform_smallint(IntType = 0, IntType = 9);
  explicit uniform_smallint(const param_type &);
  result_type a() const;
  result_type b() const;
  result_type min() const;
  result_type max() const;
  param_type param() const;
  void param(const param_type &);
  void reset();
  template<typename Engine> result_type operator()(Engine &) const;
  template<typename Engine> 
    result_type operator()(Engine &, const param_type &) const;

  // friend functions
  template<typename CharT, typename Traits> 
    std::basic_ostream< CharT, Traits > & 
    operator<<(std::basic_ostream< CharT, Traits > &, 
               const uniform_smallint &);
  template<typename CharT, typename Traits> 
    std::basic_istream< CharT, Traits > & 
    operator>>(std::basic_istream< CharT, Traits > &, 
               const uniform_smallint &);
  bool operator==(const uniform_smallint &, const uniform_smallint &);
  bool operator!=(const uniform_smallint &, const uniform_smallint &);
};

uniform_smallint 公有成员函数

  1. explicit uniform_smallint(IntType min = 0, IntType max = 9);

    构造一个 uniform_smallint. min 与 max 分别是输出范围的下界与上界.

  2. explicit uniform_smallint(const param_type & param);

    从参数构造 uniform_smallint.

  3. result_type a() const;

    返回该分布的最小值.

  4. result_type b() const;

    返回该分布的最大值.

  5. result_type min() const;

    返回该分布的最小值.

  6. result_type max() const;

    返回该分布的最大值.

  7. param_type param() const;

    返回分布的参数.

  8. void param(const param_type & param);

    设置分布的参数.

  9. void reset();

    效果: 调用 reset 之后, 分布的后续使用不依赖于任何引擎在调用 reset 之前产生的值.

  10. template<typename Engine> result_type operator()(Engine & eng) const;

    返回在 [min(), max()] 范围内均匀分布的值.

  11. template<typename Engine> 
  result_type operator()(Engine & eng, const param_type & param) const;

    返回在 [param.a(), param.b()] 范围内均匀分布的值.

uniform_smallint 友元函数

  1. template<typename CharT, typename Traits> 
  std::basic_ostream< CharT, Traits > & 
  operator<<(std::basic_ostream< CharT, Traits > & os, 
             const uniform_smallint & ud);

    把分布写入 std::ostream.

  2. template<typename CharT, typename Traits> 
  std::basic_istream< CharT, Traits > & 
  operator>>(std::basic_istream< CharT, Traits > & is, 
             const uniform_smallint & ud);

    从 std::istream 读取分布.

  3. bool operator==(const uniform_smallint & lhs, const uniform_smallint & rhs);

    如果给定相同的生成器时两个分布将产生相同的值序列, 则返回 true.

  4. bool operator!=(const uniform_smallint & lhs, const uniform_smallint & rhs);

    如果给定相同的生成器时两个分布可能产生不同的值序列, 则返回 true.
