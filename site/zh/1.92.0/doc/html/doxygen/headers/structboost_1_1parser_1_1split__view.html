结构体模板 split_view

boost::parser::split_view

概要

// In header: <boost/parser/split.hpp>

template<std::ranges::viewable_range V, typename Parser, typename GlobalState, 
         typename ErrorHandler, typename SkipParser> 
struct split_view {
  // member classes/structs/unions
  template<bool Const> 
  struct iterator {
    // types
    typedef unspecified I;        
    typedef unspecified S;        
    typedef unspecified base_type;

    // public member functions
    iterator() = default;
    iterator(unspecified);
    iterator & operator++();
    BOOST_PARSER_SUBRANGE< I > operator*() const;
  };
  template<bool Const> 
  struct sentinel {
  };

  // public member functions
  split_view() = default;
  split_view(V, parser_interface< Parser, GlobalState, ErrorHandler > const &, 
             parser_interface< SkipParser > const &, trace = trace::off);
  split_view(V, parser_interface< Parser, GlobalState, ErrorHandler > const &, 
             trace = trace::off);
  V base() const;
  V base();
  auto begin();
  auto end();
  auto begin() const;
  auto end() const;
};

说明

产生类型为 V 的底层序列的子范围序列. 底层序列被切分为以给定解析器的匹配为界的子范围, 可能使用给定的跳过解析器.

split_view 公有成员函数

  1. split_view() = default;
  2. split_view(V base, 
           parser_interface< Parser, GlobalState, ErrorHandler > const & parser, 
           parser_interface< SkipParser > const & skip, 
           trace trace_mode = trace::off);
  3. split_view(V base, 
           parser_interface< Parser, GlobalState, ErrorHandler > const & parser, 
           trace trace_mode = trace::off);
  4. V base() const;
  5. V base();
  6. auto begin();
  7. auto end();
  8. auto begin() const;
  9. auto end() const;
