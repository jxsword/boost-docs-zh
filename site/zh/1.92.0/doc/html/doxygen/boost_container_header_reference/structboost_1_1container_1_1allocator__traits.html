结构体模板 allocator_traits

boost::container::allocator_traits

概要

// In header: <boost/container/allocator_traits.hpp>

template<typename Allocator> 
struct allocator_traits {
  // types
  typedef Allocator                  allocator_type;                        
  typedef allocator_type::value_type value_type;                            
  typedef unspecified                pointer;                               
  typedef see_documentation          const_pointer;                         
  typedef see_documentation          reference;                             
  typedef see_documentation          const_reference;                       
  typedef see_documentation          void_pointer;                          
  typedef see_documentation          const_void_pointer;                    
  typedef see_documentation          difference_type;                       
  typedef see_documentation          size_type;                             
  typedef see_documentation          propagate_on_container_copy_assignment;
  typedef see_documentation          propagate_on_container_move_assignment;
  typedef see_documentation          propagate_on_container_swap;           
  typedef see_documentation          is_always_equal;                       
  typedef see_documentation          is_partially_propagable;               
  template<typename T> 
  using rebind_alloc = see_documentation;
  template<typename T> 
  using rebind_traits = allocator_traits< rebind_alloc< T > >;

  // member classes/structs/unions
  template<typename T> 
  struct portable_rebind_alloc {
    // types
    typedef see_documentation type;
  };

  // public static functions
  static pointer allocate(Allocator &, size_type);
  static void deallocate(Allocator &, pointer, size_type);
  static pointer allocate(Allocator &, size_type, const_void_pointer);
  template<typename T> static void destroy(Allocator &, T *) noexcept;
  static size_type max_size(const Allocator &) noexcept;
  static Allocator select_on_container_copy_construction(const Allocator &);
  template<typename T, class ... Args> 
    static void construct(Allocator &, T *, Args &&...);
  static bool storage_is_unpropagable(const Allocator &, pointer) noexcept;
  static bool equal(const Allocator &, const Allocator &) noexcept;
};

说明

类模板 allocator_traits 为所有分配器类型提供统一接口. 此类是 std::allocator_traits 的 C++03 兼容实现

allocator_traits 公有类型

  1. typedef unspecified pointer;

    Allocator::pointer 若存在这样的类型; 否则为 value_type*

  2. typedef see_documentation const_pointer;

    Allocator::const_pointer 若存在这样的类型; 否则为 pointer_traits<pointer>::rebind<const

  3. typedef see_documentation reference;

    非标准扩展 Allocator::reference 若存在这样的类型; 否则为 value_type&

  4. typedef see_documentation const_reference;

    非标准扩展 Allocator::const_reference 若存在这样的类型; 否则为 const value_type&

  5. typedef see_documentation void_pointer;

    Allocator::void_pointer 若存在这样的类型; 否则为 pointer_traits<pointer>::rebind<void>.

  6. typedef see_documentation const_void_pointer;

    Allocator::const_void_pointer 若存在这样的类型; 否则为 pointer_traits<pointer>::rebind<const

  7. typedef see_documentation difference_type;

    Allocator::difference_type 若存在这样的类型; 否则为 pointer_traits<pointer>::difference_type.

  8. typedef see_documentation size_type;

    Allocator::size_type 若存在这样的类型; 否则为 make_unsigned<difference_type>::type

  9. typedef see_documentation propagate_on_container_copy_assignment;

    Allocator::propagate_on_container_copy_assignment 若存在这样的类型, 否则为具有内部常量静态布尔成员 value == false 的类型.

  10. typedef see_documentation propagate_on_container_move_assignment;

    Allocator::propagate_on_container_move_assignment 若存在这样的类型, 否则为具有内部常量静态布尔成员 value == false 的类型.

  11. typedef see_documentation propagate_on_container_swap;

    Allocator::propagate_on_container_swap 若存在这样的类型, 否则为具有内部常量静态布尔成员 value == false 的类型.

  12. typedef see_documentation is_always_equal;

    Allocator::is_always_equal 若存在这样的类型, 否则为具有内部常量静态布尔成员 value == is_empty<Allocator>::value 的类型.

  13. typedef see_documentation is_partially_propagable;

    Allocator::is_partially_propagable 若存在这样的类型, 否则为具有内部常量静态布尔成员 value == false 的类型. 注释: 用于实现 small_vector_allocator 的非标准扩展.

  14. template<typename T> using rebind_alloc = see_documentation;

    定义一个分配器: Allocator::rebind<T>::other 若存在这样的类型; 否则为 Allocator<T, Args>, 若 Allocator 是形如 Allocator<U, Args> 的类模板实例化, 其中 Args 是零个或多个类型实参; 否则 rebind_alloc 的实例化非良构.

    在 C++03 编译器中 rebind_alloc 是由先前详述规则推导的分配器派生的结构体.

  15. template<typename T> using rebind_traits = allocator_traits< rebind_alloc< T > >;

    在 C++03 编译器中 rebind_traits 是从 allocator_traits<OtherAlloc> 派生的结构体, 其中 OtherAlloc 是按 rebind_alloc 中解释的规则推导的分配器.

allocator_traits 公有静态函数

  1. static pointer allocate(Allocator & a, size_type n);

    返回: a.allocate(n) 

  2. static void deallocate(Allocator & a, pointer p, size_type n);

    返回: a.deallocate(p, n)

    抛出: 无

  3. static pointer allocate(Allocator & a, size_type n, const_void_pointer p);

    效果: 若该调用良构则调用 a.allocate(n, p); 否则调用 a.allocate(n) 

  4. template<typename T> static void destroy(Allocator & a, T * p) noexcept;

    效果: 若该调用良构则调用 a.destroy(p); 否则调用 p->~T().

  5. static size_type max_size(const Allocator & a) noexcept;

    返回: 若该表达式良构则为 a.max_size(); 否则为 numeric_limits<size_type>::max().

  6. static Allocator select_on_container_copy_construction(const Allocator & a);

    返回: 若该表达式良构则为 a.select_on_container_copy_construction(); 否则为 a.

  7. template<typename T, class ... Args> 
  static void construct(Allocator & a, T * p, Args &&... args);

    效果: 若该调用良构则调用 a.construct(p, std::forward<Args>(args)...); 否则调用 placement new (static_cast<void*>(p)) T(std::forward<Args>(args)...) 

  8. static bool storage_is_unpropagable(const Allocator & a, pointer p) noexcept;

    返回: 若 is_partially_propagable::value 为 true 则为 a.storage_is_unpropagable(p); 否则为 false.

  9. static bool equal(const Allocator & a, const Allocator & b) noexcept;

    返回: true, 若 is_always_equal::value == true; 否则为 a == b.
