类模板 private_adaptive_pool

boost::container::private_adaptive_pool

概要

// In header: <boost/container/adaptive_pool.hpp>

template<typename T, std::size_t NodesPerBlock = ADP_nodes_per_block, 
         std::size_t MaxFreeBlocks = ADP_max_free_blocks, 
         std::size_t OverheadPercent = ADP_overhead_percent> 
class private_adaptive_pool {
public:
  // types
  typedef unsigned int                                                                                                               allocation_type;
  typedef private_adaptive_pool< T, NodesPerBlock, MaxFreeBlocks, OverheadPercent >                                                  self_t;         
  typedef dtl::private_adaptive_node_pool< sizeof(T), NodesPerBlock, MaxFreeBlocks, OverheadPercent, dtl::alignment_of< T >::value > pool_t;         
  typedef T                                                                                                                          value_type;     
  typedef T *                                                                                                                        pointer;        
  typedef const T *                                                                                                                  const_pointer;  
  typedef ::boost::container::dtl::unvoid_ref< T >::type                                                                             reference;      
  typedef ::boost::container::dtl::unvoid_ref< constT >::type                                                                        const_reference;
  typedef std::size_t                                                                                                                size_type;      
  typedef std::ptrdiff_t                                                                                                             difference_type;
  typedef boost::container::dtl::version_type< self_t, Version >                                                                     version;        

  // member classes/structs/unions
  template<typename T2> 
  struct rebind {
    // types
    typedef private_adaptive_pool< T2, NodesPerBlock, MaxFreeBlocks, OverheadPercent > other;
  };

  // public member functions
  private_adaptive_pool() noexcept;
  private_adaptive_pool(const private_adaptive_pool &) noexcept;
  private_adaptive_pool & operator=(const private_adaptive_pool &) noexcept;
  template<typename T2> 
    private_adaptive_pool(const private_adaptive_pool< T2, NodesPerBlock, MaxFreeBlocks, OverheadPercent > &) noexcept;
  ~private_adaptive_pool() noexcept;
  size_type max_size() const noexcept;
  pointer allocate(size_type, const void * = 0);
  void deallocate(const pointer &, size_type) noexcept;
  pointer allocation_command(allocation_type, size_type, size_type &, 
                             pointer &);
  size_type size(pointer) const noexcept;
  pointer allocate_one();
  void allocate_individual(std::size_t, multiallocation_chain &);
  void deallocate_one(pointer) noexcept;
  void deallocate_individual(multiallocation_chain &) noexcept;
  void allocate_many(size_type, std::size_t, multiallocation_chain &);
  void allocate_many(const size_type *, size_type, multiallocation_chain &);
  void deallocate_many(multiallocation_chain &) noexcept;
  void deallocate_free_blocks() noexcept;

  // friend functions
  void swap(private_adaptive_pool &, private_adaptive_pool &) noexcept;
  bool operator==(const private_adaptive_pool &, 
                  const private_adaptive_pool &) noexcept;
  bool operator!=(const private_adaptive_pool &, 
                  const private_adaptive_pool &) noexcept;

  // private member functions
  pointer priv_allocation_command(allocation_type, std::size_t, size_type &, 
                                  pointer &);

  // public data members
  BOOST_STATIC_CONSTEXPR std::size_t nodes_per_block;
  BOOST_STATIC_CONSTEXPR std::size_t max_free_blocks;
  BOOST_STATIC_CONSTEXPR std::size_t overhead_percent;
  BOOST_STATIC_CONSTEXPR std::size_t real_nodes_per_block;
  pool_t m_pool;
};

说明

private_adaptive_pool 公有类型

  1. typedef unsigned int allocation_type;

    若 Version 为 1, 该分配器是符合 STL 的分配器. 若 Version 为 2, 该分配器提供高级的就地扩展和批量分配能力.

private_adaptive_pool 公有成员函数

  1. private_adaptive_pool() noexcept;
    默认构造函数.
  2. private_adaptive_pool(const private_adaptive_pool &) noexcept;
    从另一个 private_adaptive_pool 复制构造.
  3. private_adaptive_pool & operator=(const private_adaptive_pool &) noexcept;
    从另一个 adaptive_pool 复制赋值.
  4. template<typename T2> 
  private_adaptive_pool(const private_adaptive_pool< T2, NodesPerBlock, MaxFreeBlocks, OverheadPercent > &) noexcept;
    从相关的 private_adaptive_pool 复制构造.
  5. ~private_adaptive_pool() noexcept;
    析构函数.
  6. size_type max_size() const noexcept;

    返回可分配的元素数量. 从不抛出异常

  7. pointer allocate(size_type count, const void * = 0);

    为 count 个元素的数组分配内存. 若内存不足则抛出 bad_alloc

  8. void deallocate(const pointer & ptr, size_type count) noexcept;

    释放已分配的内存. 从不抛出异常

  9. pointer allocation_command(allocation_type command, size_type limit_size, 
                           size_type & prefer_in_recvd_out_size, 
                           pointer & reuse);
  10. size_type size(pointer p) const noexcept;

    返回 p 指向的先前分配内存最多可容纳的对象数量.

  11. pointer allocate_one();

    只分配一个对象. 用此函数分配的内存只能用 deallocate_one() 释放. 若内存不足则抛出 bad_alloc

  12. void allocate_individual(std::size_t num_elements, 
                         multiallocation_chain & chain);

    分配多个大小 == 1 的元素. 这些元素必须用 deallocate_one() 逐个释放

  13. void deallocate_one(pointer p) noexcept;

    释放先前用 allocate_one() 分配的内存. 切勿使用 deallocate_one 释放用 allocate_one() 以外其他函数分配的内存. 从不抛出异常

  14. void deallocate_individual(multiallocation_chain & chain) noexcept;
  15. void allocate_many(size_type elem_size, std::size_t n_elements, 
                   multiallocation_chain & chain);

    分配多个大小为 elem_size 的元素. 这些元素必须用 deallocate() 逐个释放

  16. void allocate_many(const size_type * elem_sizes, size_type n_elements, 
                   multiallocation_chain & chain);

    分配 n_elements 个元素, 每个元素大小为 elem_sizes[i]. 这些元素必须用 deallocate() 逐个释放

  17. void deallocate_many(multiallocation_chain & chain) noexcept;
  18. void deallocate_free_blocks() noexcept;
    释放池的所有空闲块.

private_adaptive_pool 友元函数

  1. void swap(private_adaptive_pool &, private_adaptive_pool &) noexcept;

    交换分配器. 不抛出异常. 若两个分配器位于不同的内存段, 则结果未定义.

  2. bool operator==(const private_adaptive_pool &, const private_adaptive_pool &) noexcept;

    分配器之间的比较总为 true, 因为用一个实例分配的内存可以由另一个实例释放

  3. bool operator!=(const private_adaptive_pool &, const private_adaptive_pool &) noexcept;

    分配器之间的比较总为 false, 因为用一个实例分配的内存可以由另一个实例释放

private_adaptive_pool 私有成员函数

  1. pointer priv_allocation_command(allocation_type command, 
                                std::size_t limit_size, 
                                size_type & prefer_in_recvd_out_size, 
                                pointer & reuse_ptr);
