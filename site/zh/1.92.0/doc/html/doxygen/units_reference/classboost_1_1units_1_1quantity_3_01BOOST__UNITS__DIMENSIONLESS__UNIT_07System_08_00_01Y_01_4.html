类模板 quantity<BOOST_UNITS_DIMENSIONLESS_UNIT(System), Y>

boost::units::quantity<BOOST_UNITS_DIMENSIONLESS_UNIT(System), Y>

概要

// In header: <boost/units/quantity.hpp>

template<typename System, typename Y> 
class quantity<BOOST_UNITS_DIMENSIONLESS_UNIT(System), Y> {
public:
  // types
  typedef quantity< unit< dimensionless_type, System >, Y > this_type;     
  typedef Y                                                 value_type;    
  typedef System                                            system_type;   
  typedef dimensionless_type                                dimension_type;
  typedef unit< dimension_type, system_type >               unit_type;     

  // public member functions
  quantity();
  quantity(value_type);
  quantity(const this_type &);
  BOOST_CXX14_CONSTEXPR this_type & operator=(const this_type &);
  template<typename YY> 
    quantity(const quantity< unit< dimension_type, system_type >, YY > &, 
             unspecified = 0);
  template<typename YY> 
    explicit quantity(const quantity< unit< dimension_type, system_type >, YY > &, 
                      unspecified = 0);
  template<typename YY> 
    BOOST_CXX14_CONSTEXPR this_type & 
    operator=(const quantity< unit< dimension_type, system_type >, YY > &);
  template<typename System2, typename Y2> 
    quantity(const quantity< unit< dimensionless_type, System2 >, Y2 > &, 
             unspecified = 0, unspecified = 0, unspecified = 0);
  template<typename System2, typename Y2> 
    explicit quantity(const quantity< unit< dimensionless_type, System2 >, Y2 > &, 
                      unspecified = 0, unspecified = 0, unspecified = 0);
  template<typename System2, typename Y2> 
    explicit quantity(const quantity< unit< dimensionless_type, System2 >, Y2 > &, 
                      unspecified = 0);
  template<typename System2> 
    BOOST_CXX14_CONSTEXPR this_type & 
    operator=(const quantity< BOOST_UNITS_DIMENSIONLESS_UNIT(System2), Y > &);
  BOOST_CONSTEXPR operator value_type() const;
  BOOST_CONSTEXPR const value_type & value() const;
  BOOST_CXX14_CONSTEXPR this_type & operator+=(const this_type &);
  BOOST_CXX14_CONSTEXPR this_type & operator-=(const this_type &);
  BOOST_CXX14_CONSTEXPR this_type & operator*=(const value_type &);
  BOOST_CXX14_CONSTEXPR this_type & operator/=(const value_type &);

  // public static functions
  static BOOST_CONSTEXPR this_type from_value(const value_type &);

  // private member functions
   BOOST_MPL_ASSERT_NOT(unspecified);
};

说明

针对无量纲量的特化. 单位系统之间允许隐式转换, 因为所有无量纲量都是等价的. 从 value_type 隐式构造与赋值以及隐式转换到它也是允许的.

quantity 公有成员函数

  1. quantity();
  2. quantity(value_type val);
    允许从原始 value_type 构造
  3. quantity(const this_type & source);
  4. BOOST_CXX14_CONSTEXPR this_type & operator=(const this_type & source);
  5. template<typename YY> 
  quantity(const quantity< unit< dimension_type, system_type >, YY > & source, 
           unspecified = 0);
    若值类型自身允许隐式转换, 则值类型之间的隐式转换是允许的
  6. template<typename YY> 
  explicit quantity(const quantity< unit< dimension_type, system_type >, YY > & source, 
                    unspecified = 0);
    若值类型自身不允许隐式转换, 则值类型之间的隐式转换是不允许的
  7. template<typename YY> 
  BOOST_CXX14_CONSTEXPR this_type & 
  operator=(const quantity< unit< dimension_type, system_type >, YY > & source);
    若值类型自身允许隐式赋值, 则值类型之间的隐式赋值是允许的
  8. template<typename System2, typename Y2> 
  quantity(const quantity< unit< dimensionless_type, System2 >, Y2 > & source, 
           unspecified = 0, unspecified = 0, unspecified = 0);
    不同单位系统之间允许隐式转换
  9. template<typename System2, typename Y2> 
  explicit quantity(const quantity< unit< dimensionless_type, System2 >, Y2 > & source, 
                    unspecified = 0, unspecified = 0, unspecified = 0);
    不同单位系统之间允许隐式转换
  10. template<typename System2, typename Y2> 
  explicit quantity(const quantity< unit< dimensionless_type, System2 >, Y2 > & source, 
                    unspecified = 0);

    当单位不等价时, 不同单位系统之间的转换为显式.

  11. template<typename System2> 
  BOOST_CXX14_CONSTEXPR this_type & 
  operator=(const quantity< BOOST_UNITS_DIMENSIONLESS_UNIT(System2), Y > & source);
    不同单位系统之间允许隐式赋值
  12. BOOST_CONSTEXPR operator value_type() const;
    允许隐式转换为 value_type 
  13. BOOST_CONSTEXPR const value_type & value() const;
    值的常量访问器

    若 add_typeof_helper<value_type,value_type>::type 可转换为 value_type, 则可以加上相同类型的量

  14. BOOST_CXX14_CONSTEXPR this_type & operator+=(const this_type & source);
    若 subtract_typeof_helper<value_type,value_type>::type 可转换为 value_type, 则可以减去相同类型的量
  15. BOOST_CXX14_CONSTEXPR this_type & operator-=(const this_type & source);
    若 multiply_typeof_helper<value_type,value_type>::type 可转换为 value_type, 则可以把量乘以标量 value_type 
  16. BOOST_CXX14_CONSTEXPR this_type & operator*=(const value_type & val);
    若 divide_typeof_helper<value_type,value_type>::type 可转换为 value_type, 则可以把量除以标量 value_type 
  17. BOOST_CXX14_CONSTEXPR this_type & operator/=(const value_type & val);

quantity 公有静态函数

    直接从 value_type 构造量.
