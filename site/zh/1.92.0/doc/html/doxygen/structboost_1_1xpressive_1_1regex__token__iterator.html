结构体模板 regex_token_iterator

boost::xpressive::regex_token_iterator

概要

// In header: <boost/xpressive/regex_token_iterator.hpp>

template<typename BidiIter> 
struct regex_token_iterator {
  // types
  typedef basic_regex< BidiIter >          regex_type;       
  typedef iterator_value< BidiIter >::type char_type;        
  typedef sub_match< BidiIter >            value_type;       
  typedef std::ptrdiff_t                   difference_type;  
  typedef value_type const *               pointer;          
  typedef value_type const &               reference;        
  typedef std::forward_iterator_tag        iterator_category;

  // public member functions
  regex_token_iterator();
  regex_token_iterator(BidiIter, BidiIter, basic_regex< BidiIter > const &);
  template<typename LetExpr> 
    regex_token_iterator(BidiIter, BidiIter, basic_regex< BidiIter > const &, 
                         unspecified);
  template<typename Subs> 
    regex_token_iterator(BidiIter, BidiIter, basic_regex< BidiIter > const &, 
                         Subs const &, 
                         regex_constants::match_flag_type = regex_constants::match_default);
  template<typename Subs, typename LetExpr> 
    regex_token_iterator(BidiIter, BidiIter, basic_regex< BidiIter > const &, 
                         Subs const &, unspecified, 
                         regex_constants::match_flag_type = regex_constants::match_default);
  regex_token_iterator(regex_token_iterator< BidiIter > const &);
  regex_token_iterator< BidiIter > & 
  operator=(regex_token_iterator< BidiIter > const &);
  value_type const & operator*() const;
  value_type const * operator->() const;
  regex_token_iterator< BidiIter > & operator++();
  regex_token_iterator< BidiIter > operator++(int);
};

  10. regex_token_iterator< BidiIter > & operator++();

    若 N == -1, 则把 *this 设置为序列末尾迭代器. 否则, 若 N+1 < subs.size(), 则递增 N 并把 result 设置为 ((subs[N] == -1) ? value_type(what.prefix().str()) : value_type(what[subs[N]].str())). 否则, 若 what.prefix().first != what[0].second 且元素 match_prev_avail 未在 flags 中设置, 则设置它. 然后像调用 regex_search(what[0].second, end, what, *pre, flags) 那样寻找下一个匹配, 但有以下变化: 若先前找到的匹配长度为零 (what[0].length() == 0), 则尝试从 what[0].second 开始寻找非零长度的匹配; 只有在该尝试失败且 what[0].second != suffix().second 成立时, 才从 what[0].second + 1 开始寻找 (可能为零长度的) 匹配. 若找到这样的匹配, 则把 N 设置为零, 并把 result 设置为 ((subs[N] == -1) ? value_type(what.prefix().str()) : value_type(what[subs[N]].str())). 否则, 若找不到更多匹配, 令 last_end 为最后找到的匹配的结束点. 若 last_end != end 且 subs[0] == -1, 则把 N 设置为 -1, 并把 result 设置为 value_type(last_end, end). 否则把 *this 设置为序列末尾迭代器.
