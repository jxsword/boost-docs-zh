类模板 circular_list_algorithms

boost::intrusive::circular_list_algorithms

概要

// In header: <boost/intrusive/circular_list_algorithms.hpp>

template<typename NodeTraits> 
class circular_list_algorithms {
public:
  // types
  typedef NodeTraits::node           node;          
  typedef NodeTraits::node_ptr       node_ptr;      
  typedef NodeTraits::const_node_ptr const_node_ptr;
  typedef NodeTraits                 node_traits;   

  // member classes/structs/unions

  struct stable_partition_info {

    // public data members
    std::size_t num_1st_partition;
    std::size_t num_2nd_partition;
    node_ptr beg_2st_partition;
  };

  // public static functions
  static void init(node_ptr) noexcept;
  static bool inited(const_node_ptr) noexcept;
  static void init_header(node_ptr) noexcept;
  static bool is_empty(const_node_ptr) noexcept;
  static bool unique(const_node_ptr) noexcept;
  static std::size_t count(const_node_ptr) noexcept;
  static node_ptr unlink(node_ptr) noexcept;
  static void unlink(node_ptr, node_ptr) noexcept;
  static void link_before(node_ptr, node_ptr) noexcept;
  static void link_after(node_ptr, node_ptr) noexcept;
  static void swap_nodes(node_ptr, node_ptr) noexcept;
  static void transfer(node_ptr, node_ptr, node_ptr) noexcept;
  static void transfer(node_ptr, node_ptr) noexcept;
  static void reverse(node_ptr) noexcept;
  static void move_backwards(node_ptr, std::size_t) noexcept;
  static void move_forward(node_ptr, std::size_t) noexcept;
  static std::size_t distance(const_node_ptr, const_node_ptr) noexcept;
  template<typename Pred> 
    static void stable_partition(node_ptr, node_ptr, Pred, 
                                 stable_partition_info &);

  // private static functions
  static void swap_prev(node_ptr, node_ptr) noexcept;
  static void swap_next(node_ptr, node_ptr) noexcept;
};

说明

circular_list_algorithms 提供操作构成循环双向链表的节点的基本算法. 空的循环链表由一个指针指向自身的节点构成.

circular_list_algorithms 以一个 NodeTraits 类配置, 该类封装要操作的节点的信息. NodeTraits 必须支持以下接口:

Typedefs:

node: 构成循环列表的节点的类型

node_ptr: 指向节点的指针

const_node_ptr: 指向 const 节点的指针

静态函数:

static node_ptr get_previous(const_node_ptr n);

static void set_previous(node_ptr n, node_ptr prev);

static node_ptr get_next(const_node_ptr n);

static void set_next(node_ptr n, node_ptr next);

circular_list_algorithms 公有静态函数

  1. static void init(node_ptr this_node) noexcept;

    效果: 构造一个未使用的列表元素, 使得 inited(this_node) == true

    复杂度: 常数

    抛出: 无.

  2. static bool inited(const_node_ptr this_node) noexcept;

    效果: 若 "this_node" 处于如同由 "init" 函数初始化的未使用状态, 则返回 true.

    复杂度: 常数

    抛出: 无.

  3. static void init_header(node_ptr this_node) noexcept;

    效果: 构造一个空列表, 使 this_node 成为循环列表的唯一节点: NodeTraits::get_next(this_node) == NodeTraits::get_previous(this_node) == this_node.

    复杂度: 常数

    抛出: 无.

  4. static bool is_empty(const_node_ptr this_node) noexcept;

    效果: 若 this_node 指向一个空列表则返回 true.

    复杂度: 常数

    抛出: 无.

  5. static bool unique(const_node_ptr this_node) noexcept;

    要求: this_node 必须位于循环列表中或为一个空循环列表.

    效果: 若 "this_node" 是循环列表的唯一节点则返回 true: return NodeTraits::get_next(this_node) == this_node

    复杂度: 常数

    抛出: 无.

  6. static std::size_t count(const_node_ptr this_node) noexcept;

    要求: this_node 必须位于循环列表中或为一个空循环列表.

    效果: 返回循环列表中的节点数量. 若循环列表为空, 返回 1.

    复杂度: 线性

    抛出: 无.

  7. static node_ptr unlink(node_ptr this_node) noexcept;

    要求: this_node 必须位于循环列表中或为一个空循环列表.

    效果: 从循环列表中解除该节点的链接.

    复杂度: 常数

    抛出: 无.

  8. static void unlink(node_ptr b, node_ptr e) noexcept;

    要求: b 与 e 必须是同一循环列表的节点或一个空范围.

    效果: 从循环列表中解除节点范围 [b, e) 的链接.

    复杂度: 常数

    抛出: 无.

  9. static void link_before(node_ptr nxt_node, node_ptr this_node) noexcept;

    要求: nxt_node 必须是循环列表中的节点.

    效果: 在循环列表中把 this_node 链接到 nxt_node 之前.

    复杂度: 常数

    抛出: 无.

  10. static void link_after(node_ptr prev_node, node_ptr this_node) noexcept;

    要求: prev_node 必须是循环列表中的节点.

    效果: 在循环列表中把 this_node 链接到 prev_node 之后.

    复杂度: 常数

    抛出: 无.

  11. static void swap_nodes(node_ptr this_node, node_ptr other_node) noexcept;

    要求: this_node 与 other_node 必须是已插入循环列表的节点或为空循环列表.

    效果: 交换节点的位置: this_node 被插入第二个循环列表中 other_node 的位置, other_node 被插入第一个循环列表中 this_node 的位置.

    复杂度: 常数

    抛出: 无.

  12. static void transfer(node_ptr p, node_ptr b, node_ptr e) noexcept;

    要求: b 与 e 必须是同一循环列表的节点或一个空范围, 且 p 必须是另一循环列表的节点或不得是该列表中的迭代器

    效果: 把 [b, e) 范围内的节点从其循环列表中移除, 并插入到 p 的循环列表中 p 之前.

    复杂度: 常数

    抛出: 无.

  13. static void transfer(node_ptr p, node_ptr i) noexcept;

    要求: i 必须是循环列表中的节点, p 必须是另一循环列表中的节点.

    效果: 把节点 i 从其循环列表中移除, 并插入到 p 的循环列表中 p 之前. 若 p == i 或 p == NodeTraits::get_next(i), 此函数为空操作.

    复杂度: 常数

    抛出: 无.

  14. static void reverse(node_ptr p) noexcept;

    效果: 反转 list 中元素的顺序.

    抛出: 无.

    复杂度: 此函数是线性时间.

  15. static void move_backwards(node_ptr p, std::size_t n) noexcept;

    效果: 把节点 p 向列表末尾方向移动 n 个位置.

    抛出: 无.

    复杂度: 与移动的位置数成线性关系.

  16. static void move_forward(node_ptr p, std::size_t n) noexcept;

    效果: 把节点 p 向列表开头方向移动 n 个位置.

    抛出: 无.

    复杂度: 与移动的位置数成线性关系.

  17. static std::size_t distance(const_node_ptr f, const_node_ptr l) noexcept;

    要求: f 与 l 必须位于循环列表中.

    效果: 返回范围 [f, l) 中的节点数量.

    复杂度: 线性

    抛出: 无.

  18. template<typename Pred> 
  static void stable_partition(node_ptr beg, node_ptr end, Pred pred, 
                               stable_partition_info & info);

circular_list_algorithms 私有静态函数

  2. static void swap_next(node_ptr this_node, node_ptr other_node) noexcept;
