basic_socket_streambuf

套接字的 iostream streambuf.

template<
    typename Protocol,
    typename Clock = chrono::steady_clock,
    typename WaitTraits = wait_traits<Clock>>
class basic_socket_streambuf :
  public std::streambuf,
  basic_socket< Protocol >
