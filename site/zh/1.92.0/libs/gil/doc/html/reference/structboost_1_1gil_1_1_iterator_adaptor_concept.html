template<typename Iterator>
struct boost::gil::IteratorAdaptorConcept< Iterator >

迭代器适配器是适配另一个前向迭代器的前向迭代器.

除 GIL 迭代器要求外, GIL 迭代器适配器还必须提供以下元函数:

  • is_iterator_adaptor<Iterator>: 返回 std::true_type
  • iterator_adaptor_get_base<Iterator>: 返回基迭代器类型
  • iterator_adaptor_rebind<Iterator,NewBase>: 用新迭代器替换基迭代器

被适配对象可通过迭代器的 "base()" 方法获得.

concept IteratorAdaptorConcept<boost_concepts::ForwardTraversalConcept Iterator>
{
where SameType<is_iterator_adaptor<Iterator>::type, std::true_type>;
typename iterator_adaptor_get_base<Iterator>;
where Metafunction<iterator_adaptor_get_base<Iterator> >;
where boost_concepts::ForwardTraversalConcept<iterator_adaptor_get_base<Iterator>::type>;
typename another_iterator;
typename iterator_adaptor_rebind<Iterator,another_iterator>::type;
where boost_concepts::ForwardTraversalConcept<another_iterator>;
where IteratorAdaptorConcept<iterator_adaptor_rebind<Iterator,another_iterator>::type>;
const iterator_adaptor_get_base<Iterator>::type& Iterator::base() const;
};
