详细语义 - 可选引用

constexpr optional<T&>::optional() noexcept;

constexpr optional<T&>::optional(none_t) noexcept;

  • 后置条件: bool(*this) == false; *this 不引用任何东西.

template<class R> optional<T&>::optional(R&& r) noexcept;

  • 后置条件: bool(*this) == true; addressof(**this) == addressof(r).
  • 备注: 除非 R 是左值引用, 否则程序非良构. 若 decay<R> 是 boost::optional 的实例, 此构造函数不参与重载决议.
  • 注释: 在不能正确支持绑定到整型类型的 const 左值的编译器上, 此构造函数被声明为 explicit. 详情 参见此处.

  • 示例: 

    T v;
T& vref = v ;
optional<T&> opt(vref);
assert ( *opt == v ) ;
++ v ; // mutate referee
assert (*opt == v);

template<class R> optional<T&>::optional(bool cond, R&& r) noexcept;

  • 效果: 初始化 ref, 所用表达式为 cond ? addressof(r) : nullptr.
  • 后置条件: bool(*this) == cond; 若 bool(*this), 则 addressof(**this) == addressof(r).
  • 备注: 除非 R 是左值引用, 否则程序非良构. 若 decay<R> 是 boost::optional 的实例, 此构造函数不参与重载决议.

optional<T&>::optional ( optional const& rhs ) noexcept ;

  • 效果: 初始化 ref, 所用表达式为 rhs.ref.
  • 后置条件: bool(*this) == bool(rhs).

  • 示例: 

    optional<T&> uninit ;
assert (!uninit);

optional<T&> uinit2 ( uninit ) ;
assert ( uninit2 == uninit );

T v = 2 ; T& ref = v ;
optional<T> init(ref);
assert ( *init == v ) ;

optional<T> init2 ( init ) ;
assert ( *init2 == v ) ;

v = 3 ;

assert ( *init  == 3 ) ;
assert ( *init2 == 3 ) ;

template<class U> explicit optional<T&>::optional ( optional<U&> const& rhs ) noexcept ;

  • 要求: is_convertible<U&, T&>::value 是 true.
  • 效果: 初始化 ref, 所用表达式为 rhs.ref.
  • 后置条件: bool(*this) == bool(rhs).

optional<T&>::operator= ( none_t ) noexcept ;

  • 效果: 把 ref 用表达式 nullptr 赋值.
  • 返回: *this.
  • 后置条件: bool(*this) == false.

optional& optional<T&>::operator= ( optional const& rhs ) noexcept ;

  • 效果: 把 ref 用表达式 rhs.ref 赋值.
  • 返回: *this.
  • 后置条件: bool(*this) == bool(rhs).
  • 注释: 该行为被称为 重绑定语义. 详情见 此处.

  • 示例: 

    int a = 1 ;
int b = 2 ;
T& ra = a ;
T& rb = b ;
optional<int&> def ;
optional<int&> ora(ra) ;
optional<int&> orb(rb) ;

def = orb ; // binds 'def' to 'b' through 'rb' wrapped within 'orb'
assert ( *def == b ) ;
*def = ora ; // changes the value of 'b' to a copy of the value of 'a'
assert ( b == a ) ;
int c = 3;
int& rc = c ;
optional<int&> orc(rc) ;
ora = orc ; // REBINDS ora to 'c' through 'rc'
c = 4 ;
assert ( *ora == 4 ) ;

template<class U> optional& optional<T&>::operator= ( optional<U&> const& rhs ) noexcept ;

  • 要求: is_convertible<U&, T&>::value 是 true.
  • 效果: 把 ref 用表达式 rhs.ref 赋值.
  • 返回: *this.
  • 后置条件: bool(*this) == bool(rhs).

template<class R> optional& optional<T&>::operator= ( R&& r ) noexcept ;

  • 效果: 把 ref 用表达式 r 赋值.
  • 返回: *this.
  • 后置条件: bool(*this) == true.
  • 备注: 除非 R 是左值引用, 否则程序非良构. 若 decay<R> 是 boost::optional 的实例, 此函数不参与重载决议.

  • 示例: 

    int a = 1 ;
int b = 2 ;
T& ra = a ;
T& rb = b ;
optional<int&> def ;
optional<int&> opt(ra) ;

def = rb ; // binds 'def' to 'b' through 'rb'
assert ( *def == b ) ;
*def = a ; // changes the value of 'b' to a copy of the value of 'a'
assert ( b == a ) ;
int c = 3;
int& rc = c ;
opt = rc ; // REBINDS to 'c' through 'rc'
c = 4 ;
assert ( *opt == 4 ) ;

void optional<T&>::emplace( R&& r ) noexcept ;

  • 效果: 把 ref 用表达式 r 赋值.
  • 后置条件: bool(*this) == true.
  • 备注: 除非 R 是左值引用, 否则程序非良构. 若 decay<R> 是 boost::optional 的实例, 此函数不参与重载决议.

T& optional<T&>::get() const ;

T& optional<T&>::operator *() const ;

  • 要求: bool(*this) == true.
  • 效果: 返回 *ref.
  • 抛出: 无.

  • 示例: 

    T v ;
T& vref = v ;
optional<T&> opt ( vref );
T const& vref2 = *opt;
assert ( vref2 == v ) ;
++ v ;
assert ( *opt == v ) ;

T* optional<T&>::operator -> () const ;

  • 要求: bool(*this) == true.
  • 效果: 返回 ref.
  • 抛出: 无.

T& optional<T&>::value() const ;

  • Effects: Equivalent to return bool(*this) ? *val : throw bad_optional_access();.

template<class R> T& optional<T&>::value_or( R&& r ) const noexcept;

  • Effects: Equivalent to if (*this) return **this; else return r;.
  • 备注: 除非 R 是左值引用, 否则程序非良构.

template<class F> T& optional<T&>::value_or( F f ) const ;

  • Effects: Equivalent to if (*this) return **this; else return f();.
  • 备注: 除非 decltype(f()) 是左值引用, 否则程序非良构.

template<class F> auto optional<T&>::map( F f ) const -> 见下文;

  • 效果: 等价于 if (*this) return f(**this); else return none;.
  • 备注: 该函数的返回类型是 optional<decltype(f(**this))>.

template<class F> auto optional<T&>::flat_map( F f ) const -> 见下文;

  • 要求: 表达式 f(**this) 的返回类型是 optional<U>, 对某个对象或引用类型 U 而言.
  • 效果: 等价于 if (*this) return f(**this); else return none;.
  • 备注: 该函数的返回类型是 optional<U>.

T* optional<T&>::get_ptr () const noexcept;

  • 返回: ref.

bool has_value() const noexcept;

optional<T&>::operator bool () const noexcept;

  • 返回: bool(ref).

optional<T&>::operator ! () const noexcept;

  • 返回: !bool(ref).

void optional<T&>::reset() noexcept;

  • 效果: 同 *this = none.

template<class R> void optional<T&>::reset ( R&& r) noexcept;

  • 效果: 等价于 *this = std::forward<R>(r).
  • 备注: 该函数已被弃用.

bool optional<T&>::is_initialized() const noexcept;

  • 效果: 等价于 return bool(*this).
  • 备注: 该函数已被弃用.

template<class R> T& optional<T&>::get_value_or( R&& r ) const noexcept;

  • 效果: 等价于 return value_or(std::forward<R>(r);.
  • 备注: 该函数已被弃用.
