data

概述
参考

作者

  • Glen Fernandes

概述

头文件 <boost/core/data.hpp> 提供函数模板 data, 用于获得指向范围中第一个元素的指针.

参考 

namespace boost {

template<class C>
constexpr auto
data(C& c) noexcept(noexcept(c.data())) -> decltype(c.data());

template<class C>
constexpr auto
data(const C& c) noexcept(noexcept(c.data())) -> decltype(c.data());

template<class T, std::size_t N>
constexpr T*
data(T(&a)[N]) noexcept;

template<class T>
constexpr const T*
data(std::initializer_list<T> l) noexcept;

} /* boost */

函数

template<class C> constexpr auto data(C& c) noexcept(noexcept(c.data())) -> decltype(c.data());

返回 c.data().

template<class C> constexpr auto data(const C& c) noexcept(noexcept(c.data())) -> decltype(c.data());

返回 c.data().

template<class T, std::size_t N> constexpr T* data(T(&a)[N]) noexcept;

返回 a.

template<class T> constexpr const T* data(std::initializer_list<T> l) noexcept;

返回 l.begin().
