#include <boost/polygon/polygon.hpp>
#include <cassert>
namespace gtl = boost::polygon;
using namespace boost::polygon::operators;

int main() {
//让我们给自己声明一个多边形集合
using namespace gtl; //为了使用运算符
typedef std::vector<polygon_data<int> > PolygonSet;
PolygonSet ps;

//放入一些数据
ps += rectangle_data<int>(0, 0, 10, 10);

//现在做点有趣的事
PolygonSet ps2;
ps2 += rectangle_data<int>(5, 5, 15, 15);
PolygonSet ps3;
assign(ps3, ps * ps2); //哇, 我刚才感觉房间在我周围弯折了
PolygonSet ps4;
ps4 += ps + ps2;

//断言结果面积等于
//输入几何面积之和减去输入之间重叠的面积
assert(area(ps4) == area(ps) + area(ps2) - area(ps3));

//我已经看不到代码了, 看到的全是
//包围盒……区间……三角形

//让我们用慢动作再来一遍, 好吗?
assert(equivalence((ps + ps2) - (ps * ps2), ps ^ ps2));

//嗯, 从并集中减去交集
//等价于异或, 而这一切只用一行代码,
//现在我们进入了子弹时间编程
//(顺便说一句, xor 是单遍实现的, 而非组合)

//纯粹为了好玩
rectangle_data<int> rect;
assert(extents(rect, ps ^ ps2));
assert(area(rect) == 225);
assert(area(rect ^ (ps ^ ps2)) == area(rect) - area(ps ^ ps2));
return 0;
}

//现在你知道了如何在库多边形上使用多边形集合概念
