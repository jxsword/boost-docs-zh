#include <boost/polygon/polygon.hpp>
#include <cassert>
namespace gtl = boost::polygon;
using namespace boost::polygon::operators;

//让我们把 point_usage.cpp 中 main 的主体改写
//为按点类型参数化的泛型函数
template <typename Point>
void test_point() {
//构造一个 gtl 点
int x = 10;
int y = 20;
//Point pt(x, y);
Point pt = gtl::construct<Point>(x, y);
assert(gtl::x(pt) == 10);
assert(gtl::y(pt) == 20);

//各向同性点访问速成指南
typedef gtl::orientation_2d O;
using gtl::HORIZONTAL;
using gtl::VERTICAL;
O o = HORIZONTAL;
assert(gtl::x(pt) == gtl::get(pt, o));

o = o.get_perpendicular();
assert(o == VERTICAL);
assert(gtl::y(pt) == gtl::get(pt, o));

gtl::set(pt, o, 30);
assert(gtl::y(pt) == 30);

//使用一些库函数
//Point pt2(10, 30);
Point pt2 = gtl::construct<Point>(10, 30);
assert(gtl::equivalence(pt, pt2));

gtl::transformation<int> tr(gtl::axis_transformation::SWAP_XY);
gtl::transform(pt, tr);
assert(gtl::equivalence(pt, gtl::construct<Point>(30, 10)));

gtl::transformation<int> tr2 = tr.inverse();
assert(tr == tr2); //SWAP_XY 的逆变换是其自身

gtl::transform(pt, tr2);
assert(gtl::equivalence(pt, pt2)); //两个点再次相等

gtl::move(pt, o, 10); //沿 y 方向移动 pt 10 个单位
assert(gtl::euclidean_distance(pt, pt2) == 10.0f);

gtl::move(pt, o.get_perpendicular(), 10); //沿 x 方向移动 pt 10 个单位
assert(gtl::manhattan_distance(pt, pt2) == 20);
}

//现在让我们声明自己的点类型
//Bjarne 说过, 如果一个类不维护
//不变式, 就直接用 struct.
struct CPoint {
int x;
int y;
};

//就这样, 简单又漂亮……但等等, 它什么都做不了
//我们如何用它完成一个点需要做的所有事情?


//首先我们将其注册为 boost polygon 的点
namespace boost { namespace polygon {
template <>
struct geometry_concept<CPoint> { typedef point_concept type; };


//然后为我们的点类型特化 gtl 点特征
template <>
struct point_traits<CPoint> {
typedef int coordinate_type;

static inline coordinate_type get(const CPoint& point,
orientation_2d orient) {
if(orient == HORIZONTAL)
return point.x;
return point.y;
}
};

template <>
struct point_mutable_traits<CPoint> {
typedef int coordinate_type;

static inline void set(CPoint& point, orientation_2d orient, int value) {
if(orient == HORIZONTAL)
point.x = value;
else
point.y = value;
}
static inline CPoint construct(int x_value, int y_value) {
CPoint retval;
retval.x = x_value;
retval.y = y_value;
return retval;
}
};
} }

//现在让我们看看 CPoint 能否与库函数配合使用
int main() {
test_point<CPoint>(); //太好了! 所有测试都替你完成了.
return 0;
}
