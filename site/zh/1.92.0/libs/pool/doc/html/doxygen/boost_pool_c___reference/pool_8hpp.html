头文件 <boost/pool/pool.hpp>

提供了类 pool: 一个快速的内存分配器, 保证所有已分配内存块的正确对齐, 并对简单分隔存储方案提供的框架进行扩展与泛化. 还提供了两个可与 pool 配合使用的 UserAllocator 类.

namespace boost {
  struct default_user_allocator_malloc_free;
  struct default_user_allocator_new_delete;

  template<typename UserAllocator> class pool;

  pool< UserAllocator >::size_type const pool< UserAllocator >::min_alloc_size;
  pool< UserAllocator >::size_type const pool< UserAllocator >::min_align;
}
