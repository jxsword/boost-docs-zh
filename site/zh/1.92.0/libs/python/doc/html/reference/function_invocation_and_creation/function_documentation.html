boost/python/function_doc_signature.hpp

引言
类 function_doc_signature_generator
Example

引言

Boost.Python 支持自动附加 Python 式与 C++ 签名的文档字符串. 该特性由类 function_doc_signature_generator 实现. 该类使用全部重载、所提供的参数名与默认值, 以及用户定义的文档字符串, 为给定的函数生成文档.

类 function_doc_signature_generator

该类只有一个公有函数, 返回一个记录函数各重载的字符串列表.

namespace boost { namespace python { namespace objects {

    class function_doc_signature_generator
    {
      public:
          static list function_doc_signatures(function const *f);
    };

}}}

Example 

#include <boost/python/module.hpp>
#include <boost/python/def.hpp>
#include <boost/python/args.hpp>
#include <boost/python/tuple.hpp>
#include <boost/python/class.hpp>
#include <boost/python/overloads.hpp>
#include <boost/python/raw_function.hpp>

using namespace boost::python;

tuple f(int x = 1, double y = 4.25, char const* z = "wow")
{
    return make_tuple(x, y, z);
}

BOOST_PYTHON_FUNCTION_OVERLOADS(f_overloads, f, 0, 3)


struct X
{
    tuple f(int x = 1, double y = 4.25, char const* z = "wow")
    {
        return make_tuple(x, y, z);
    }
};

BOOST_PYTHON_MEMBER_FUNCTION_OVERLOADS(X_f_overloads, X::f, 0, 3)

tuple raw_func(tuple args, dict kw)
{
    return make_tuple(args, kw);
}

BOOST_PYTHON_MODULE(args_ext)
{
    def("f", f, (arg("x")=1, arg("y")=4.25, arg("z")="wow")
        , "This is f's docstring"
        );

    def("raw", raw_function(raw_func));

    def("f1", f, f_overloads("f1's docstring", args("x", "y", "z")));


    class_<X>("X", "This is X's docstring", init<>(args("self")))
        .def("f", &X::f
             , "This is X.f's docstring"
             , args("self","x", "y", "z"))

        ;

}

boost/python/pytype_function.hpp

引言
类 wrap_pytype
类 registered_pytype
类 expected_from_python_type
类 to_python_target_type
Example

引言

为支持 Python 式签名, 转换器应提供一个 get_pytype 函数, 返回指向关联 PyTypeObject 的指针. 例如参见 ResultConverter 或 to_python_converter. 本头文件中的类旨在实现 get_pytype 时使用. 还有 _direct 版本的 class T 模板, 它们应与未修饰的类型参数一起使用, 并期望在模块加载时该类型已在转换注册表中.

类 wrap_pytype

本模板生成一个静态 get_pytype 成员, 返回该模板参数.

namespace boost { namespace python { namespace converter{

    template < PyTypeObject const *pytype >
    class wrap_pytype
    {
      public:
          static PyTypeObject const *get_pytype(){return pytype; }
    };

}}}

类 registered_pytype

本模板应与 (可能被修饰的) 用 class_ 导出到 Python 的类型模板参数一起使用. 所生成的静态 get_pytype 成员返回相应的 Python 类型.

namespace boost { namespace python { namespace converter{

    template < class T >
    class registered_pytype
    {
      public:
          static PyTypeObject const *get_pytype();
    };

}}}

类 expected_from_python_type

本模板生成一个静态 get_pytype 成员, 它检查 from_python 中为类型 T 注册的转换器, 并返回匹配的 Python 类型.

namespace boost { namespace python { namespace converter{

    template < class T >
    class expected_from_python_type
    {
      public:
          static PyTypeObject const *get_pytype();
    };

}}}

类 to_python_target_type

本模板生成一个静态 get_pytype 成员, 返回 T 可转换到的 Python 类型.

namespace boost { namespace python { namespace converter{

    template < class T >
    class to_python_target_type
    {
      public:
          static PyTypeObject const *get_pytype();
    };

}}}

Example

本示例假定有人已经实现了 Python 文档中的标准 noddy 示例模块, 并把相应的声明放在 "noddy.h" 中. 由于 noddy_NoddyObject 是最简单不过的扩展类型, 这个示例有点刻意: 它包装了一个函数, 其全部信息都包含在返回值的类型中.

C++ 模块定义:

#include <boost/python/reference.hpp>
#include <boost/python/module.hpp>
#include "noddy.h"

struct tag {};
tag make_tag() { return tag(); }

using namespace boost::python;

struct tag_to_noddy
#if defined BOOST_PYTHON_SUPPORTS_PY_SIGNATURES //unnecessary overhead if py signatures are not supported
: wrap_pytype<&noddy_NoddyType> //inherits get_pytype from wrap_pytype
#endif
{
    static PyObject* convert(tag const& x)
    {
        return PyObject_New(noddy_NoddyObject, &noddy_NoddyType);
    }
};

BOOST_PYTHON_MODULE(to_python_converter)
{
    def("make_tag", make_tag);
    to_python_converter<tag, tag_to_noddy
#if defined BOOST_PYTHON_SUPPORTS_PY_SIGNATURES //invalid if py signatures are not supported
          , true
#endif
          >(); //"true" because tag_to_noddy has member get_pytype
}

下面的示例使用模板 expected_from_python_type 和 to_pyhton_target_type 注册到 Python 和从 Python 的转换器.

#include <boost/python/module.hpp>
#include <boost/python/def.hpp>
#include <boost/python/extract.hpp>
#include <boost/python/to_python_converter.hpp>
#include <boost/python/class.hpp>

using namespace boost::python;

struct A
{
};

struct B
{
  A a;
  B(const A& a_):a(a_){}
};

// Converter from A to python int
struct BToPython
#if defined BOOST_PYTHON_SUPPORTS_PY_SIGNATURES //unnecessary overhead if py signatures are not supported
   : converter::to_python_target_type<A>  //inherits get_pytype
#endif
{
  static PyObject* convert(const B& b)
  {
    return incref(object(b.a).ptr());
  }
};

// Conversion from python int to A
struct BFromPython
{
  BFromPython()
  {
    boost::python::converter::registry::push_back
        ( &convertible
        , &construct
        , type_id< B >()
#if defined BOOST_PYTHON_SUPPORTS_PY_SIGNATURES //invalid if py signatures are not supported
        , &converter::expected_from_python_type<A>::get_pytype//convertible to A can be converted to B
#endif
        );
  }

  static void* convertible(PyObject* obj_ptr)
  {
      extract<const A&> ex(obj_ptr);
      if (!ex.check()) return 0;
      return obj_ptr;
  }

  static void construct(
      PyObject* obj_ptr,
      converter::rvalue_from_python_stage1_data* data)
  {
    void* storage = (
        (converter::rvalue_from_python_storage< B >*)data)-> storage.bytes;

    extract<const A&> ex(obj_ptr);
    new (storage) B(ex());
    data->convertible = storage;
  }
};


B func(const B& b) { return b ; }

BOOST_PYTHON_MODULE(pytype_function_ext)
{
  to_python_converter< B , BToPython
#if defined BOOST_PYTHON_SUPPORTS_PY_SIGNATURES //invalid if py signatures are not supported
             ,true
#endif
             >(); //has get_pytype
  BFromPython();

  class_<A>("A") ;

  def("func", &func);

}
