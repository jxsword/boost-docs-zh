handle 是指向 Python 对象类型的智能指针; 它持有一个 T* 类型的指针, 其中 T 是它的模板参数. T 必须要么是派生自 PyObject 的类型, 要么是一个 POD 类型, 其最初 sizeof(PyObject) 字节与 PyObject 布局兼容. 请在 Python/'C' API 与高层代码的边界处使用 handle<>; 对于 Python 对象的泛化接口, 更推荐使用 object.

在本文档中, 术语 "向上转型" (upcast) 指把指针 Y* 转换为基类 pointer T* 的操作——经由 static_cast<T*>, 当 Y 派生自 T 时; 或经由 C 风格转换 (T*), 当并非如此时. 然而, 在后一种情形, 如果最初 sizeof(PyObject) 字节 (属于 Y) 与 PyObject 不布局兼容, 该 "向上转型" 是病构的.

namespace boost { namespace python
{
  template <class T>
  class handle
  {
      typedef unspecified-member-function-pointer bool_type;

   public: // types
      typedef T element_type;

   public: // member functions
      ~handle();

      template <class Y>
      explicit handle(detail::borrowed<null_ok<Y> >* p);

      template <class Y>
      explicit handle(null_ok<detail::borrowed<Y> >* p);

      template <class Y>
      explicit handle(detail::borrowed<Y>* p);

      template <class Y>
      explicit handle(null_ok<Y>* p);

      template <class Y>
      explicit handle(Y* p);

      handle();

      handle& operator=(handle const& r);

      template<typename Y>
      handle& operator=(handle<Y> const & r); // never throws


      template <typename Y>
      handle(handle<Y> const& r);

      handle(handle const& r);

      T* operator-> () const;
      T& operator* () const;
      T* get() const;
      void reset();
      T* release();

      operator bool_type() const; // never throws
   private:
      T* m_p;
  };

  template <class T> struct null_ok;
  namespace detail { template <class T> struct borrowed; }
}}
