概要

// In header: <boost/log/expressions/predicates/has_attr.hpp>

template<typename T> 
class has_attribute {
public:
  // types
  typedef bool result_type;  // Function result_type. 
  typedef T    value_type;   // Expected attribute value type. 

  // public member functions
