cpp_bin_float

#include <boost/multiprecision/cpp_bin_float.hpp> 

namespace boost{ namespace multiprecision{

enum digit_base_type
{
   digit_base_2 = 2,
   digit_base_10 = 10
};

template <unsigned Digits, digit_base_type base = digit_base_10, class Allocator = void, class Exponent = int, ExponentMin = 0, ExponentMax = 0>
class cpp_bin_float;

typedef number<cpp_bin_float<50> > cpp_bin_float_50;
typedef number<cpp_bin_float<100> > cpp_bin_float_100;

typedef number<backends::cpp_bin_float<24, backends::digit_base_2, void, std::int16_t, -126, 127>, et_off>         cpp_bin_float_single;
typedef number<backends::cpp_bin_float<53, backends::digit_base_2, void, std::int16_t, -1022, 1023>, et_off>       cpp_bin_float_double;
typedef number<backends::cpp_bin_float<64, backends::digit_base_2, void, std::int16_t, -16382, 16383>, et_off>     cpp_bin_float_double_extended;
typedef number<backends::cpp_bin_float<113, backends::digit_base_2, void, std::int16_t, -16382, 16383>, et_off>    cpp_bin_float_quad;
typedef number<backends::cpp_bin_float<237, backends::digit_base_2, void, std::int32_t, -262142, 262143>, et_off>  cpp_bin_float_oct;

}} // namespaces

cpp_bin_float 后端与number 配合使用: 它充当一个纯 C++(仅头文件且无依赖)的浮点数类型, 可直接替换原生 C++ 浮点数类型, 但精度高得多.

类型cpp_bin_float 可以定点精度使用(只需指定非零的Digits 模板参数). 类型定义cpp_bin_float_50 和cpp_bin_float_100 分别提供 50 位和 100 位十进制精度的算术类型.

此外, 你还可以指定精度以十进制位数还是二进制位表示 - 例如要声明一个与cpp_bin_float 精度完全相同的double, 应使用number<cpp_bin_float<53, digit_base_2> >. 类型定义cpp_bin_float_single,cpp_bin_float_double,cpp_bin_float_quad,cpp_bin_float_oct 和cpp_bin_float_double_extended 分别提供 IEEE 单精度、双精度、四精度和八精度浮点数据类型以及 Intel 扩展双精度类型的软件对应物. 注意, 虽然这些类型在功能上等同于原生 IEEE 类型, 但它们的大小或位布局与真正的 IEEE 兼容类型并不相同.

通常cpp_bin_float 不分配内存: 其数字所需的全部空间都直接在类内分配. 因此应当注意不要以过高的位数使用该类, 否则栈空间需求可能失控. 若这构成问题, 则把分配器作为模板参数传入会使cpp_bin_float 动态分配所需内存: 这会显著减小cpp_bin_float 的大小, 并提高可行位数上限, 但以性能为代价. 不过请记住, 随着位数增长, 算术运算的开销会迅速变得非常 高: 目前的实现确实没有针对大位数进行优化或设计. 注意, 由于所分配对象的实际类型是完全不透明的, 建议使用带有void value_type 的分配器, 例如:number<cpp_bin_float<1000, digit_base_10, std::allocator<void> > >.

最后的模板参数决定指数的类型和范围: 参数Exponent 可以是任何有符号整数类型, 但注意MinExponent 和MaxExponent 不能直接取到Exponent 类型的极限, 因为内部计算需要一些额外的余量. 若出现这种情况, 你会得到一个编译期错误. 此外, 若 MinExponent 或 MaxExponent 为零, 则库会在类型约束和内部计算额外余量需求的前提下, 选择尽可能大的合适值.

此类型有完整的标准库和numeric_limits 支持.

使用此类型时你应当知道的事项:

  • 默认构造的cpp_bin_float 的值为零.
  • 此类型的基数为 2, 即使精度以十进制位数指定.
  • 此类型同时支持无穷大和 NaN. 只要结果会溢出就会生成无穷大, 任何数学上未定义的运算都会生成 NaN.
  • 此类型有std::numeric_limits 特化.
  • 任何基于此类型实例化的number 都可以转换为基于此类型实例化的其他number - 例如你可以从number<cpp_bin_float<50> > 转换为number<cpp_bin_float<SomeOtherValue> >. 收窄转换按就近舍入进行, 且是explicit.
  • 若字符串无法解释为有效的浮点数, 则从字符串转换会抛出std::runtime_error.
  • 所有算术运算都正确地就近舍入. 字符串转换和sqrt 函数也是正确舍入的, 但超越函数(sin、cos、pow、exp 等)则不是.
cpp_bin_float 示例: 
#include <boost/multiprecision/cpp_bin_float.hpp>
#include <boost/math/special_functions/gamma.hpp>

#include <iostream>

int main()
{
   using namespace boost::multiprecision;

   // Operations at fixed precision and full numeric_limits support:
   cpp_bin_float_100 b = 2;
   std::cout << std::numeric_limits<cpp_bin_float_100>::digits << std::endl;
   std::cout << std::numeric_limits<cpp_bin_float_100>::digits10 << std::endl;

   // We can use any C++ std lib function, lets print all the digits as well:
   std::cout << std::setprecision(std::numeric_limits<cpp_bin_float_100>::max_digits10)
      << log(b) << std::endl; // print log(2)

   // We can also use any function from Boost.Math:
   std::cout << boost::math::tgamma(b) << std::endl;
   // These even work when the argument is an expression template:
   std::cout << boost::math::tgamma(b * b) << std::endl;

   // And since we have an extended exponent range we can generate some really large
   // numbers here (4.0238726007709377354370243e+2564):
   std::cout << boost::math::tgamma(cpp_bin_float_100(1000)) << std::endl;
   return 0;
}
