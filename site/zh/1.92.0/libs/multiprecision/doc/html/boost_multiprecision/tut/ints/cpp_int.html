cpp_int

#include <boost/multiprecision/cpp_int.hpp> 

namespace boost{ namespace multiprecision{

typedef unspecified-type limb_type;

enum cpp_integer_type    { signed_magnitude, unsigned_magnitude };
enum cpp_int_check_type  { checked, unchecked };

template <std::size_t MinBits = 0,
          std::size_t MaxBits = 0,
          cpp_integer_type SignType = signed_magnitude,
          cpp_int_check_type Checked = unchecked,
          class Allocator = std::allocator<limb_type> >
class cpp_int_backend;
//
// Expression templates default to et_off if there is no allocator:
//
template <std::size_t MinBits, std::size_t MaxBits, cpp_integer_type SignType, cpp_int_check_type Checked>
struct expression_template_default<cpp_int_backend<MinBits, MaxBits, SignType, Checked, void> >
{ static const expression_template_option value = et_off; };

typedef number<cpp_int_backend<> >              cpp_int;    // arbitrary precision integer
typedef rational_adaptor<cpp_int_backend<> >    cpp_rational_backend;
typedef number<cpp_rational_backend>            cpp_rational; // arbitrary precision rational number

// Fixed precision unsigned types:
typedef number<cpp_int_backend<128, 128, unsigned_magnitude, unchecked, void> >   uint128_t;
typedef number<cpp_int_backend<256, 256, unsigned_magnitude, unchecked, void> >   uint256_t;
typedef number<cpp_int_backend<512, 512, unsigned_magnitude, unchecked, void> >   uint512_t;
typedef number<cpp_int_backend<1024, 1024, unsigned_magnitude, unchecked, void> > uint1024_t;

// Fixed precision signed types:
typedef number<cpp_int_backend<128, 128, signed_magnitude, unchecked, void> >     int128_t;
typedef number<cpp_int_backend<256, 256, signed_magnitude, unchecked, void> >     int256_t;
typedef number<cpp_int_backend<512, 512, signed_magnitude, unchecked, void> >     int512_t;
typedef number<cpp_int_backend<1024, 1024, signed_magnitude, unchecked, void> >   int1024_t;

// Over again, but with checking enabled this time:
typedef number<cpp_int_backend<0, 0, signed_magnitude, checked> >                 checked_cpp_int;
typedef rational_adaptor<cpp_int_backend<0, 0, signed_magnitude, checked> >       checked_cpp_rational_backend;
typedef number<cpp_rational_backend>                                              checked_cpp_rational;

// Checked fixed precision unsigned types:
typedef number<cpp_int_backend<128, 128, unsigned_magnitude, checked, void> >     checked_uint128_t;
typedef number<cpp_int_backend<256, 256, unsigned_magnitude, checked, void> >     checked_uint256_t;
typedef number<cpp_int_backend<512, 512, unsigned_magnitude, checked, void> >     checked_uint512_t;
typedef number<cpp_int_backend<1024, 1024, unsigned_magnitude, checked, void> >   checked_uint1024_t;

// Fixed precision signed types:
typedef number<cpp_int_backend<128, 128, signed_magnitude, checked, void> >       checked_int128_t;
typedef number<cpp_int_backend<256, 256, signed_magnitude, checked, void> >       checked_int256_t;
typedef number<cpp_int_backend<512, 512, signed_magnitude, checked, void> >       checked_int512_t;
typedef number<cpp_int_backend<1024, 1024, signed_magnitude, checked, void> >     checked_int1024_t;

}} // namespaces

cpp_int_backend 类型通常通过上面给出的某个便捷 typedef 使用.

此后端是整数类型中的"瑞士军刀", 因为它既能表示定点精度整数类型, 又能表示任意精度整数类型, 而且同时支持有符号和无符号类型. 它有五个模板参数:

MinBits

决定在求助于动态内存分配之前直接存储在对象内部的位数. 为零时, 此字段根据与动态存储头联合后可容纳的位数自动确定: 设置较大的值可能提升性能, 因为更大的整数值会在需要内存分配之前先存储在内部.

MaxBits

决定该类型可存储的最大位数: 由此得到定点精度类型. 当此值与 MinBits 相同时, Allocator 参数将被忽略, 因为永远不会执行动态内存分配: 此时 Allocator 参数应设置为类型void. 注意, 不应仅仅为了防止大的内存分配而使用此参数 - 该职责由分配器来承担更合适, 而且定点精度整数往往比预期更频繁地分配全部 MaxBits 的存储.

SignType

决定结果类型是否有符号. 注意, 对于任意精度类型, 此参数必须为signed_magnitude. 对于定点精度类型, 此参数可以是signed_magnitude 或unsigned_magnitude.

Checked

此参数有两个取值:checked 或unchecked. 见下文.

Allocator

用于动态内存分配的分配器; 当 MaxBits == MinBits 时为类型void.

当模板参数 Checked 设置为checked 时, 结果是一个带检查整数. 带检查整数与不带检查整数具有以下属性:

条件

带检查整数

不带检查整数

定点精度算术中的数值溢出

抛出std::overflow_error.

执行模 2MaxBits 算术.

用无法在目标类型中表示的值构造整数

抛出std::range_error.

把值按模 2MaxBits 转换; 从有符号到无符号的转换会提取输入值二进制补码表示的最后 MaxBits 位.

无符号减法得到负值

抛出std::range_error.

产生的值如同把无符号类型当作二进制补码有符号类型处理所得.

对负值尝试按位运算

抛出std::range_error

产生的值(而非位模式)如同在二进制补码整数类型上执行该运算所得.
注释

当因使用带检查类型而抛出异常时, 捕获异常后目标操作数的值是未指定的.

使用此类型时你应当知道的事项:

  • 默认构造的cpp_int_backend 的值为零.
  • 除以零会导致抛出std::overflow_error.
  • 从包含无效非数字字符的字符串构造会导致抛出std::runtime_error.
  • 由于当 allocator 参数为 void 时cpp_int_backend 的精度必然有限, 使用此类型时应注意避免数值溢出, 除非你确实想要回绕算术行为.
  • 该类型在内部使用符号-数值表示, 因此类型int128_t 具有 128 位精度外加一个额外的符号位. 在这方面, 这些类型的行为与基本(内建)二进制补码类型不同. 人们可能想改用 127 位类型, 这确实可行, 但其行为仍与二进制补码的基本(内建)类型略有不同: 其最小值和最大值(除符号外)相同, 而真正的二进制补码类型二者相差一. 应当注意, 自 C++20 起, 要求基本(内建)类型采用二进制补码, 因此这是内部表示上的细微差别.
  • 尝试把负值打印为八进制或十六进制字符串会导致抛出std::runtime_error. 这是符号-数值表示的直接后果.
  • 定点精度类型[checked_][u]intXXX_t 的表达式模板支持是关闭的 - 无论开与否, 对这些类型的性能似乎影响甚微 - 所以干脆关闭该特性以获得更快的编译速度.
  • 无符号类型支持减法 - 只要它们不是带检查整数(见上文), 结果就"如同"执行了二进制补码运算. 换句话说, 在这种情况下它们的行为与基本(内建)整数类型非常接近. 例如, 若我们使用uint128_t, 那么uint128_t(1)-4 会得到值0xFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFD(类型为uint128_t). 不过, 若此运算在checked_uint128_t 上执行, 则会抛出std::range_error.
  • 对无符号类型做一元取负会导致编译器错误(静态断言).
  • 此后端支持右值引用并具有移动感知能力, 使得基于此后端实例化的number 也具有移动感知能力.
  • 以定点精度使用时, 此类型的大小总是比你对 N 位整数的预期多一个机器字(外加编译器施加的对齐填充): 多出的那个字同时存储符号以及整数中实际在用的机器字数量. 后者是对较大定点精度整数的一种优化, 使得 1024 位整数具有与 128 位整数几乎相同的性能特征, 而不是加法慢 4 倍、乘法慢 16 倍(假定所涉的值总能放进 128 位). 通常这意味着你可以使用一个为"最坏情况"留足宽度的整数类型, 即使大多数时候算术本可以用更窄的类型完成, 性能损失也很小. 另请注意, 小到能放进最大原生整数的无符号定点精度类型会变成对该类型的简单封装, 这也包括"带检查"变体. 小的有符号类型总会有一个额外的符号字, 因此比其原生对应类型更大.
  • 当以定点精度使用且 MaxBits 小于最大原生整数类型的位数时,cpp_int_backend 会在内部切换到"简单"实现, 即仅作为单个整数的薄封装. 注意, 它仍会比裸的原生整数稍慢, 因为它模拟符号-数值表示, 而不是直接使用平台的原生符号表示: 这确保了 cpp_int 增大时行为不会发生阶跃式变化.
  • 定点精度的cpp_int 对constexpr 值和用户自定义字面量提供了一些支持, 完整说明见此处. 例如0xfffff_cppi1024 指定一个值为 0xffff 的 1024 位整数. 这可用于生成太大而无法放进任何基本(内建)数字类型的编译期常量.
  • cpp_int 类型支持 constexpr 算术, 前提是它是无分配器的定点精度类型. 它也可以是带检查的整数: 此时溢出或未定义行为会产生编译器错误. 此外, 自由函数abs、swap、multiply、add、subtract、divide_qr、integer_modulus、powm、lsb、msb、bit_test、bit_set、bit_unset、bit_flip、sqrt、gcd、lcm 等都受支持. 以这种方式使用cpp_int 需要 C++2a 编译器(即支持std::is_constant_evaluated() 的编译器), 或 C++14 模式下的 GCC-6 及更高版本. 非 GCC 且不支持std::is_constant_evaluated() 的编译器只支持非常有限的运算集: 要有心理准备会很容易碰壁.
  • 你可以把cpp_int 的原始位经由import_bits 和export_bits 函数导入/导出到外部存储. 更多信息见导入/导出一节.
示例: 
#include <boost/multiprecision/cpp_int.hpp>
#include <iostream>

int main()
{
   using namespace boost::multiprecision;

   int128_t v = 1;

   // Do some fixed precision arithmetic:
   for(unsigned i = 1; i <= 20; ++i)
      v *= i;

   std::cout << v << std::endl; // prints 2432902008176640000 (i.e. 20!)

   // Repeat at arbitrary precision:
   cpp_int u = 1;
   for(unsigned i = 1; i <= 100; ++i)
      u *= i;

   // prints 93326215443944152681699238856266700490715968264381621468592963895217599993229915608941463976156518286253697920827223758251185210916864000000000000000000000000 (i.e. 100!)
   std::cout << u << std::endl;

   return 0;
}
