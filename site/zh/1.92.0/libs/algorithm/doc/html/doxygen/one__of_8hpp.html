头文件 < boost/algorithm/cxx11/one_of.hpp >

测试范围, 查看是否恰好只有一个元素匹配某个值或谓词.

Marshall Clow 

namespace boost {
  namespace algorithm {
    template<typename InputIterator, typename Predicate> 
      BOOST_CXX14_CONSTEXPR bool 
      one_of(InputIterator, InputIterator, Predicate);
    template<typename Range, typename Predicate> 
      BOOST_CXX14_CONSTEXPR bool one_of(const Range &, Predicate);
    template<typename InputIterator, typename V> 
      BOOST_CXX14_CONSTEXPR bool 
      one_of_equal(InputIterator, InputIterator, const V &);
    template<typename Range, typename V> 
      BOOST_CXX14_CONSTEXPR bool one_of_equal(const Range &, const V &);
  }
}
